def solution_func(binary_numbers):
    # Chuyển chuỗi đầu vào thành một danh sách các số nguyên
    numbers = [int(number, 2) for number in binary_numbers.split(",")]
    
    # Tìm tất cả các số chia hết cho 5
    divisible_by_five = [number for number in numbers if number % 5 == 0]
    
    # Chuyển danh sách số thành một chuỗi nhị phân, được phân tách bởi dấu phẩy
    result = ','.join(format(number, 'b') for number in divisible_by_five)
    
    return result
