def compress_sensitive_data(data: list, threshold: int) -> list:
    if not data: return []
    
    result = []
    current_group = [data[0]]
    for i in range(1, len(data)):
        # If the signs of consecutive numbers are the same, add to the group
        if (current_group[0] >= 0) == (data[i] >= 0):
            current_group.append(data[i])
        else:  # Otherwise, start a new group
            result.extend(current_group)
            current_group = [data[i]]
    
    result.extend(current_group)  # Add the last group to the result
    
    for i in range(len(result)):
        if abs(result[i]) > threshold:
            avg = sum(result[i:i+threshold]) // len(result[i:i+threshold])
            result[i:i+threshold] = [avg]  # Replace the group with its average
    
    return result
