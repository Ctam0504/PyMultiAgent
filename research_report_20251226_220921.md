## 🏆 Giải pháp đạt điểm hội tụ cao nhất
```python
import re

def solution_func(log_string: str) -> dict:
    if not log_string:
        return {}
    
    pattern = r'ID: ([A-Za-z]), TEMP: (\d+\.\d+)'
    matches = re.findall(pattern, log_string)
    
    result = {}
    for match in matches:
        id_val = match[0]
        temp_val = float(match[1])
        
        if temp_val > 38.0:
            status = "CRITICAL"
        else:
            status = "NORMAL"
        
        result[id_val] = status
    
    return result
