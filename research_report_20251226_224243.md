def aggregate_sales(data: list) -> dict:
    result = {}
    
    for record in data:
        store = record['store']
        items = record['items']
        
        if not items:
            continue
        
        total_price = 0.0
        for item in items:
            price = item.get('price', '0')
            try:
                price = float(price)
            except ValueError:
                price = 0.0
            
            total_price += price
        
        result[store] = round(total_price, 2)
    
    return result
