def track_user_sessions(logs: list) -> dict:
    session_times = {}
    stack = []
    
    for log in logs:
        user, action, time = log['user'], log['action'], log['time']
        
        if action == 'start':
            stack.append((user, time))
        elif action == 'stop' and stack and stack[-1][0] == user:
            start_time = stack.pop()
            hour, minute = map(int, time.split(':'))
            start_hour, start_minute = map(int, start_time[1].split(':'))
            duration = (hour - start_hour) * 60 + (minute - start_minute)
            
            if user in session_times:
                session_times[user] += duration
            else:
                session_times[user] = duration
        elif action == 'stop' and stack and stack[-1][0] != user:
            continue  # Skip this stop event as it doesn't match the current start
    
    return session_times
