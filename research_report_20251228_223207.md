import re

def solution_func(data):
    # Sửa lỗi: [Mã nguồn cũ gây ra ValueError do không kiểm tra đầy đủ dữ liệu đầu vào]
    # Giải pháp: Thêm kiểm tra đầu vào là một dict và kiểm tra key 'status' có giá trị 'error' hay không.
    
    if not isinstance(data, dict):
        return "INVALID_INPUT"
    
    def is_system_critical(d):
        if isinstance(d, dict) and "status" in d:
            if d["status"] == "error":
                return True
        return False
    
    def flatten_dict(d, parent_key='', sep='.'):
        items = []
        for k, v in d.items():
            new_key = f"{parent_key}{sep}{k}" if parent_key else k
            if isinstance(v, dict):
                items.extend(flatten_dict(v, new_key, sep=sep).items())
            elif isinstance(v, float):
                items.append((new_key, round(v, 2)))
            elif isinstance(v, str):
                items.append((new_key, v.strip()))
            else:
                items.append((new_key, v))
        return dict(items)
    
    if is_system_critical(data):
        return "SYSTEM_CRITICAL"
    
    flattened_data = flatten_dict(data)
    return flattened_data
