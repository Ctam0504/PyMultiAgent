def calculate_portfolio_performance(transactions):
    if not transactions:
        return {'portfolio_value': 0.0, 'top_asset': None}
    
    portfolio = {}
    
    for transaction in transactions:
        symbol = transaction['symbol']
        amount = transaction['amount']
        price = transaction['price']
        action = transaction['type']
        
        if action == 'buy':
            if symbol not in portfolio:
                portfolio[symbol] = {'total_cost': 0.0, 'total_shares': 0.0}
            portfolio[symbol]['total_cost'] += amount * price
            portfolio[symbol]['total_shares'] += amount
        elif action == 'sell':
            if symbol in portfolio and portfolio[symbol]['total_shares'] >= amount:
                portfolio[symbol]['total_cost'] -= amount * price
                portfolio[symbol]['total_shares'] -= amount
    
    for symbol in list(portfolio.keys()):
        if portfolio[symbol]['total_shares'] == 0:
            del portfolio[symbol]
    
    total_value = sum([entry['total_cost'] for entry in portfolio.values()])
    top_asset = max(portfolio, key=lambda x: portfolio[x]['total_cost'], default=None)
    
    return {'portfolio_value': round(total_value, 2), 'top_asset': top_asset}
