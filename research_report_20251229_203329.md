def process_fuel_logs(missions):
    results = []
    
    for mission in missions:
        if mission['fuel_type'] == 'PLASMA':
            # Check if amount is an even integer
            if isinstance(mission['amount'], int) and mission['amount'] % 2 == 0:
                actual_amount = mission['amount'] * 0.8 if mission['status'] == 'EMERGENCY' else mission['amount']
                if actual_amount < 100:
                    results.append('LOW_FUEL')
                else:
                    results.append('STABLE')
            else:
                continue  # Skip this mission as it doesn't meet the criteria
        else:
            # For non-PLASMA fuel types, check if status is valid and calculate accordingly
            if mission['status'] == 'EMERGENCY':
                actual_amount = mission['amount'] * 0.8
            elif mission['status'] == 'NORMAL':
                actual_amount = mission['amount']
            else:
                continue  # Skip this mission as it has an invalid status
            
            if actual_amount < 100:
                results.append('LOW_FUEL')
            else:
                results.append('STABLE')
    
    return results
