def format_document_structure(raw_text):
    lines = raw_text.split('\n')
    stack = []  # Stack to keep track of the current hierarchy level
    result = []  # List to store the final structure
    
    for line in lines:
        if line.startswith('#'):
            depth = len(line) - line.lstrip('#').find('#')  # Determine the header level
            title = line[depth:].strip()  # Extract the title content
            
            while stack and stack[-1]['level'] >= depth:
                stack.pop()  # Close the current header if it's not the highest level
            
            new_entry = {'title': title, 'content': [], 'children': []}
            if stack:
                if stack[-1]['level'] == depth - 1:
                    stack[-1]['children'].append(new_entry)
                else:
                    raise ValueError("Invalid header level sequence")
            else:
                result.append(new_entry)
            
            stack.append({'title': title, 'content': [], 'children': [], 'level': depth})
        else:
            if stack:
                stack[-1]['content'].append(line.strip())
    
    return result
