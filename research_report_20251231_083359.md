def manage_inventory(current_stock, transactions, restock_threshold):
    final_stock = current_stock.copy()
    restock_list = []
    errors = []
    
    for transaction in transactions:
        item, action, quantity = transaction.split(' | ')
        quantity = int(quantity)
        
        if action == "IN":
            final_stock[item] = final_stock.get(item, 0) + quantity
        elif action == "OUT":
            if final_stock.get(item, 0) >= quantity:
                final_stock[item] -= quantity
            else:
                errors.append({"item": item, "action": action, "quantity": quantity, "status": "failed"})
    
    for item, stock in final_stock.items():
        if stock <= restock_threshold:
            restock_list.append(item)
    
    return {"final_stock": final_stock, "restock_list": sorted(restock_list), "errors": errors}

# Sửa lỗi: Lỗi logic trong việc xử lý giao dịch OUT và cập nhật tồn kho.
# Giải pháp: Dùng hàm .get() để không bị KeyError khi item chưa từng xuất hiện trong final_stock, sử dụng 2 vòng lặp riêng biệt cho việc cập nhật tồn kho và ghi nhận lỗi.
